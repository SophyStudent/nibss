// ==========================================
// FUNCTION 1: THE FACTORY (Higher-Order Function)
// ==========================================
const catchAsync = (fn) => {
//                  └─── INPUT BASKET 1: Holds whatever function you pass in (e.g., getBalance)

  // ------------------------------------------
  // FUNCTION 2: THE RETURNED MIDDLEWARE
  // ------------------------------------------
  return (req, res, next) => {
//│      └─── INPUT BASKET 2: Express provides these 3 objects when a user hits the route
//└── RETURN 1: Spits out Function 2 directly to Express

    // TRIGGER 1: Pulls the trigger on the function stored in Basket 1 (fn)
    // PASSING ARGUMENTS: Hands req, res, next directly into fn
    fn(req, res, next)
      
      // ------------------------------------------
      // FUNCTION 3: THE ERROR CATCHER (Callback)
      // ------------------------------------------
      .catch((err) => next(err));
//           │        └── TRIGGER 3: Hands the error over to Express error handling
//           └── INPUT BASKET 3: Catches the error object if fn fails
  };
};

// ==========================================
// FUNCTION 4: YOUR CONTROLLER (Async Function)
// ==========================================
const getBalance = async (req, res, next) => {
//                        └── INPUT BASKET 4: Receives the 3 objects handed to it by TRIGGER 1
  
  // TRIGGER 2: Calls the database and waits for the result
  const balance = await fetchFromDatabase();

  // Network output: Sends the data over the wire to the frontend
  res.json({ balance });
};

// ==========================================
// THE CONNECTION (Passing Function 4 into Function 1)
// ==========================================
app.get("/balance", catchAsync(getBalance));
//                             └── ARGUMENT: getBalance passed WITHOUT () as a package of instructions
